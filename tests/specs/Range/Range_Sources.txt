== should format a string source ==
{
 "cells": [
  {
   "cell_type": "markdown",
   "source": "code [|block|] 1"
  },
  {
   "cell_type": "markdown",
   "source": "code block 2"
  }
 ]
}

[expect]
{
 "cells": [
  {
   "cell_type": "markdown",
   "source": "code [|block|] 1_markdown"
  },
  {
   "cell_type": "markdown",
   "source": "code block 2"
  }
 ]
}

    "line 4"
   ]
  },
  {
   "cell_type": "markdown",
   "source": [
    "line 1\n",
    "line 2"
   ]
  }
 ]
}

[expect]
{
 "cells": [
  {
   "cell_type": "markdown",
   "source": [
    "line 1\n",
    "li[|ne|] 2\n",
    "\n",
    "line 4_markdown"
   ]
  },
  {
   "cell_type": "markdown",
   "source": [
    "line 1\n",
    "line 2"
   ]
  }
 ]
}

 ]
}
