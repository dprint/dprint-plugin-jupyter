== should only format the cell the range is in ==
{
 "cells": [
  {
   "cell_type": "markdown",
   "metadata": {},
   "source": [
    "code block 1"
   ]
  },
  {
   "cell_type": "code",
   "metadata": {},
   "source": [
    "code [|block|] 2"
   ]
  },
  {
   "cell_type": "markdown",
   "metadata": {},
   "source": [
    "code block 3"
   ]
  }
 ],
 "metadata": {
  "language_info": {
   "name": "python"
  }
 }
}

[expect]
{
 "cells": [
  {
   "cell_type": "markdown",
   "metadata": {},
   "source": [
    "code block 1"
   ]
  },
  {
   "cell_type": "code",
   "metadata": {},
   "source": [
    "code [|block|] 2_python"
   ]
  },
  {
   "cell_type": "markdown",
   "metadata": {},
   "source": [
    "code block 3"
   ]
  }
 ],
 "metadata": {
  "language_info": {
   "name": "python"
  }
 }
}
