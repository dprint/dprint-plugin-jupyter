== should format a code cell with the notebook language ==
{
 "cells": [
  {
   "cell_type": "code",
   "source": "code [||]block 1"
  },
  {
   "cell_type": "code",
   "source": "code block 2"
  }
 ],
 "metadata": {
  "language_info": {
   "name": "python"
  }
 }
}

[expect]
{
 "cells": [
  {
   "cell_type": "code",
   "source": "code [||]block 1_python"
  },
  {
   "cell_type": "code",
   "source": "code block 2"
  }
 ],
 "metadata": {
  "language_info": {
   "name": "python"
  }
 }
}
