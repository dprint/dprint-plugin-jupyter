== should not change anything for a range in the notebook metadata ==
{
 "cells": [
  {
   "cell_type": "code",
   "source": "code block 1"
  }
 ],
 "metadata": {
  "language_info": {
   [|"name": "python"|]
  }
 }
}

[expect]
{
 "cells": [
  {
   "cell_type": "code",
   "source": "code block 1"
  }
 ],
 "metadata": {
  "language_info": {
   [|"name": "python"|]
  }
 }
}

== should not change anything for a range in the notebook format properties ==
{
 "cells": [
  {
   "cell_type": "markdown",
   "source": "code block 1"
  }
 ],
 "nbformat": [|4,
 "nbformat_minor": 2|]
}

[expect]
{
 "cells": [
  {
   "cell_type": "markdown",
   "source": "code block 1"
  }
 ],
 "nbformat": [|4,
 "nbformat_minor": 2|]
}
